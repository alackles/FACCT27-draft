% --- Narrowed text environment ---
% Frees ~25% of \textwidth on the right for annotation notes.
\newenvironment{annotpage}{%
  \begin{list}{}{%
    \setlength{\leftmargin}{0pt}%
    \setlength{\rightmargin}{0.25\textwidth}%
    \setlength{\parsep}{0.5\baselineskip}%
    \setlength{\itemsep}{0pt}%
  }%
  \item\relax
}{%
  \end{list}%
}

% --- Margin geometry: notes go in the column annotpage frees ---
% Notes are ordinary \marginpar's, so LaTeX stacks them instead of
% overprinting; marginfix carries overflow to the next page. The paper
% has no other \marginpar, so the geometry is set globally (it is read
% at shipout, after annotpage's group may have closed).
\usepackage{marginfix}
\makeatletter
\AtBeginDocument{%
  \setlength{\marginparwidth}{0.22\textwidth}%
  \setlength{\marginparsep}{\dimexpr-0.25\textwidth+1em\relax}%
  \setlength{\marginparpush}{0.5\baselineskip}%
  \@mparswitchfalse % twoside would put even-page notes on the left
  \normalmarginpar
}
\makeatother

% --- Inline annotation command ---
% #1 = marker (number or short label)
% #2 = highlighted text span
% #3 = margin annotation text; if blank, only the superscript is set
\newcommand{\annot}[3]{%
  \hl{#2}\textsuperscript{\small #1}%
  \ifblank{#3}{}{%
    \marginpar{%
      % \normalcolor: soul leaves the highlight color open at the end of
      % a line where a highlight wraps, and the note would inherit it.
      \normalcolor\footnotesize\RaggedRight #1.~#3%
    }%
  }%
}
